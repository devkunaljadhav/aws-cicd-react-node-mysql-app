#!/bin/bash
# Debug sathi: sagla output ya file madhe jato
exec > >(tee /var/log/user-data.log | logger -t user-data -s 2>/dev/console) 2>&1
set -x

# 1. Install CodeDeploy agent FIRST (ap-south-1 region)
yum install -y ruby wget
cd /home/ec2-user
wget https://aws-codedeploy-ap-south-1.s3.ap-south-1.amazonaws.com/latest/install
chmod +x ./install
./install auto
systemctl enable --now codedeploy-agent

# 2. Update packages
yum update -y

# 3. Install Node.js
curl -fsSL https://rpm.nodesource.com/setup_20.x | bash -
yum install -y nodejs

# 4. Install PM2 globally
npm install -g pm2

# 5. Log directory and files
LOG_DIR="/var/log/react-node-mysql-app/backend"
mkdir -p $LOG_DIR
touch $LOG_DIR/combined.log
touch $LOG_DIR/error.log
chown -R root:root $LOG_DIR

# 6. PM2 startup on reboot
env PATH=$PATH:/usr/bin pm2 startup systemd -u root --hp /root
pm2 save

# 7. Install CloudWatch agent
yum install -y amazon-cloudwatch-agent

# 8. CloudWatch agent configuration
tee /opt/aws/amazon-cloudwatch-agent/etc/amazon-cloudwatch-agent.json > /dev/null <<'EOL'
{
  "logs": {
    "logs_collected": {
      "files": {
        "collect_list": [
          {
            "file_path": "/var/log/react-node-mysql-app/backend/combined.log",
            "log_group_name": "node-app-logs-backend",
            "log_stream_name": "{instance_id}-combined-log",
            "timestamp_format": "%Y-%m-%d %H:%M:%S"
          },
          {
            "file_path": "/var/log/react-node-mysql-app/backend/error.log",
            "log_group_name": "node-app-logs-backend",
            "log_stream_name": "{instance_id}-error-log",
            "timestamp_format": "%Y-%m-%d %H:%M:%S"
          },
          {
            "file_path": "/var/log/aws/codedeploy-agent/codedeploy-agent.log",
            "log_group_name": "codedeploy-agent-logs-backend",
            "log_stream_name": "{instance_id}-agent-log"
          },
          {
            "file_path": "/opt/codedeploy-agent/deployment-root/deployment-logs/codedeploy-agent-deployments.log",
            "log_group_name": "codedeploy-agent-logs-backend",
            "log_stream_name": "{instance_id}-codedeploy-agent-deployment-log"
          },
          {
            "file_path": "/tmp/codedeploy-agent.update.log",
            "log_group_name": "codedeploy-agent-logs-backend",
            "log_stream_name": "{instance_id}-codedeploy-agent-updater-log"
          }
        ]
      }
    }
  }
}
EOL

# 9. Start CloudWatch agent
/opt/aws/amazon-cloudwatch-agent/bin/amazon-cloudwatch-agent-ctl -a fetch-config -m ec2 -c file:/opt/aws/amazon-cloudwatch-agent/etc/amazon-cloudwatch-agent.json -s
